# ==============================================================================
# Stage 1: Build jdansak/Canvas-Scraper Go CLI Binary
# ==============================================================================
FROM golang:1.23-bookworm AS go-builder

RUN git clone --depth 1 https://github.com/jdansak/Canvas-Scraper.git /src/canvas-scraper
WORKDIR /src/canvas-scraper

RUN CGO_ENABLED=0 GOOS=linux go build -ldflags="-s -w" -o /out/canvas-scraper .


# ==============================================================================
# Stage 2: Build Python Environment with uv
# ==============================================================================
FROM python:3.12-slim-bookworm AS py-builder

COPY --from=ghcr.io/astral-sh/uv:0.5.11 /uv /uvx /bin/

ENV UV_COMPILE_BYTECODE=1 \
    UV_LINK_MODE=copy

WORKDIR /app

# Install dependencies first for optimal Docker layer caching
COPY pyproject.toml uv.lock ./
RUN uv sync --frozen --no-dev --no-install-project

# Copy application source and install project into .venv
COPY src/ ./src/
RUN uv sync --frozen --no-dev


# ==============================================================================
# Stage 3: Minimal Non-Root Runtime Image (Python + Playwright + Go Binary)
# ==============================================================================
FROM python:3.12-slim-bookworm AS runtime

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PATH="/app/.venv/bin:/usr/local/bin:$PATH" \
    PLAYWRIGHT_BROWSERS_PATH="/opt/pw-browsers"

# Install minimal runtime utilities + Playwright Chromium & OS dependencies
RUN apt-get update && apt-get install -y --no-install-recommends \
    ca-certificates \
    tzdata \
    && rm -rf /var/lib/apt/lists/*

# Create non-root user matching standard Ubuntu UID/GID 1000
RUN groupadd --gid 1000 appuser \
    && useradd --uid 1000 --gid 1000 --create-home --shell /bin/bash appuser

WORKDIR /app

# Copy compiled Canvas-Scraper Go binary and Python virtual environment
COPY --from=go-builder /out/canvas-scraper /usr/local/bin/canvas-scraper
COPY --from=py-builder --chown=appuser:appuser /app/.venv /app/.venv
COPY --from=py-builder --chown=appuser:appuser /app/src /app/src

# Bake Playwright Chromium + system libraries into /opt/pw-browsers before dropping root
RUN mkdir -p /opt/pw-browsers /app/config /app/secrets /app/data \
    && /app/.venv/bin/playwright install --with-deps chromium \
    && chown -R appuser:appuser /opt/pw-browsers /app \
    && rm -rf /var/lib/apt/lists/*

USER appuser

# Verify both binaries and core config directory exist
HEALTHCHECK --interval=60s --timeout=10s --start-period=15s --retries=3 \
    CMD python -c "import pathlib, sys; sys.exit(0 if pathlib.Path('/app/config/students.yaml').exists() else 1)"

ENTRYPOINT ["python", "-m", "canvas_tracker.main"]
CMD ["--config", "/app/config/students.yaml"]
